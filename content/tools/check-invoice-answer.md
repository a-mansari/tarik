{"readable": bool, "extracted": {"invoice_number": str|null, "invoice_date_raw": str|null, "invoice_date_language": "fr"|"en", "amount_untaxed": num|null, "amount_tva": num|null, "amount_stamp": num|null, "amount_disbursement": num|null, "amount_guarantee_holdback": num|null, "amount_prorata": num|null, "amount_insurance": num|null, "amount_ts": num|null, "amount_tsp": num|null, "advance_amount": num|null, "amount_total": num|null, "tier": str|null, "designation": str|null, "ice": str|null, "vat": str|null, "is_credit_note": bool, "currency": str|null, "payment_method": str|null, "period_start": str|null, "period_end": str|null, "country": str|null, "tva_lines": [{"rate": num, "amount_ht": num, "amount_tva": num}]}, "bad_fields": [{"field": str, "stored": str, "extracted": str, "problem": str}]}
